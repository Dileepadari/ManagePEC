def AvgStudents():
    try:
        query = """
                    SELECT AVG(CountStudents) AS AverageEnrollment
                    FROM (
                        SELECT S.Sport_ID, COUNT(St.Student_ID) AS CountStudents
                        FROM SPORTS AS S
                        LEFT JOIN STUDENT_SPORT_DETAILS AS St ON S.Sport_ID = St.Assigned_Sport
                        GROUP BY S.Sport_ID
                    ) AS Subquery;
                """

        cur.execute(query)
        result_tuples = cur.fetchall()
        for row in result_tuples:
            print(tuple(row.values()))
        con.commit()
    
    except Exception as e:
        con.rollback()
        print(f"Failed to retrieve with average students")
        print("Error:", e)
